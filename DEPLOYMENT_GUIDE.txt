1. SUMMARY OF WHAT WAS ADDED & MODIFIED
================================================================================

A. NEW FILES CREATED:
   1. backend/Dockerfile
      - Builds a lightweight, secure Node.js 20 Alpine container running as a non-root user.
   2. backend/.dockerignore
      - Excludes node_modules, logs, and sensitive files from the Docker build context.
   3. frontend/Dockerfile
      - Multi-stage build: compiles React/Vite into static assets, then embeds the custom Nginx configuration directly inside the container to serve the app and reverse-proxy `/api/` requests to the backend.
   4. frontend/.dockerignore
      - Excludes node_modules, dist, and local environment files.
   5. docker-compose.yml
      - Orchestrates 3 isolated containers:
        - `mern-mongo`: Official MongoDB 7.0 with persistent storage volume `mongo_data`.
        - `mern-backend`: Express API on internal port 8000, linked to MongoDB.
        - `mern-frontend`: Nginx + React listening on host port 80.
        All 3 connect via a secure private bridge network `mern-network`.
   6. aws/deploy.sh
      - Shell script executed on the EC2 server that pulls the latest code from GitHub `main`, rebuilds containers cleanly, and prunes unused Docker images to save disk space.
   7. .github/workflows/deploy.yml
      - GitHub Actions workflow triggered automatically on pushes to `main`. It connects to EC2 via SSH and runs `aws/deploy.sh`. It automatically defaults the SSH user to "ubuntu".

B. EXISTING FILES UPDATED:
   1. backend/src/connectToDb.js
      - Added support for `process.env.MONGO_URI` so it connects to `mongodb://mongo:27017/eduhub` inside Docker while still defaulting to `localhost` for local dev.
   2. backend/index.js
      - Added support for `process.env.PORT` and flexible `CORS` origin handling.

2.3. File: frontend/Dockerfile (Embedded Nginx)
--------------------------------------------------------------------------------
FROM node:20-alpine AS build

WORKDIR /app

COPY package*.json ./
RUN npm ci

COPY . .
RUN npm run build

FROM nginx:alpine

RUN rm -rf /usr/share/nginx/html/*
COPY --from=build /app/dist /usr/share/nginx/html

RUN cat <<'EOF' > /etc/nginx/conf.d/default.conf
server {
    listen 80;
    server_name localhost;

    root /usr/share/nginx/html;
    index index.html index.htm;

    gzip on;
    gzip_vary on;
    gzip_min_length 10240;
    gzip_proxied expired no-cache no-store private auth;
    gzip_types text/plain text/css text/xml text/javascript application/x-javascript application/xml application/javascript;
    gzip_disable "MSIE [1-6]\.";

    # Reverse proxy API requests to backend container
    location /api/ {
        proxy_pass http://backend:8000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;
        proxy_read_timeout 90;
    }

    # Handle React Router (SPA)
    location / {
        try_files $uri $uri/ /index.html;
    }

    # Cache static assets
    location ~* \.(?:ico|css|js|gif|jpe?g|png|woff2?|eot|ttf|svg)$ {
        expires 6M;
        access_log off;
        add_header Cache-Control "public, max-age=15552000, immutable";
    }

    error_page 500 502 503 504 /50x.html;
    location = /50x.html {
        root /usr/share/nginx/html;
    }
}
EOF

EXPOSE 80

CMD ["nginx", "-g", "daemon off;"]

2.5. File: docker-compose.yml
--------------------------------------------------------------------------------
services:
  # 1. MongoDB Database Container
  mongo:
    image: mongo:7.0
    container_name: mern-mongo
    restart: unless-stopped
    volumes:
      - mongo_data:/data/db
    networks:
      - mern-network
    healthcheck:
      test: ["CMD", "mongosh", "--eval", "db.adminCommand('ping')"]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 10s

  # 2. Node.js / Express Backend Container
  backend:
    build:
      context: ./backend
      dockerfile: Dockerfile
    container_name: mern-backend
    restart: unless-stopped
    environment:
      - PORT=8000
      - MONGO_URI=mongodb://mongo:27017/eduhub
      - NODE_ENV=production
      - JWT_SECRET=${JWT_SECRET:-eduhub_default_production_secret_key_2026}
      - CLIENT_URL=${CLIENT_URL:-*}
    depends_on:
      mongo:
        condition: service_healthy
    networks:
      - mern-network

  # 3. React + Nginx Frontend Container
  frontend:
    build:
      context: ./frontend
      dockerfile: Dockerfile
    container_name: mern-frontend
    restart: unless-stopped
    ports:
      - "80:80"
    depends_on:
      - backend
    networks:
      - mern-network

volumes:
  mongo_data:
    driver: local

networks:
  mern-network:
    driver: bridge

2.6. File: aws/deploy.sh
--------------------------------------------------------------------------------
#!/usr/bin/env bash
set -e

echo "=========================================================="
echo "🚀 Starting Automated Deployment: $(date)"
echo "=========================================================="

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
PROJECT_ROOT="$(dirname "$SCRIPT_DIR")"

cd "$PROJECT_ROOT"
echo "📂 Working directory: $(pwd)"

# Check available disk space (Target: 20GB storage on t3.medium)
echo "📊 Current Disk Space:"
df -h /

echo "📥 Pulling latest changes from Git..."
git fetch origin main
git reset --hard origin/main

if [ ! -f .env ]; then
  echo "ℹ️  No root .env file found. Using default environment variables in docker-compose.yml."
fi

echo "📦 Pulling updated base container images..."
docker compose pull mongo || true

echo "🔨 Building images and recreating containers..."
docker compose up -d --build --remove-orphans

echo "🧹 Pruning unused Docker images and build caches..."
docker image prune -f

echo "🔍 Checking container status..."
docker compose ps

echo "=========================================================="
echo "✅ Deployment completed successfully!"
echo "🌐 Application is live on port 80."
echo "=========================================================="

2.7. File: .github/workflows/deploy.yml
--------------------------------------------------------------------------------
name: Deploy to AWS EC2

on:
  push:
    branches:
      - main
  workflow_dispatch:

jobs:
  deploy:
    name: Continuous Deployment (EC2)
    runs-on: ubuntu-latest

    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Execute Remote Deployment via SSH
        uses: appleboy/ssh-action@v1.0.3
        with:
          host: ${{ secrets.EC2_HOST }}
          username: ${{ secrets.EC2_USER || 'ubuntu' }}
          key: ${{ secrets.EC2_SSH_KEY }}
          port: 22
          script_stop: true
          command_timeout: 30m
          script: |
            echo "Connected to EC2 host: ${{ secrets.EC2_HOST }}"
            if [ -d "$HOME/MERN_Project" ]; then
              TARGET_DIR="$HOME/MERN_Project"
            else
              TARGET_DIR="$(pwd)"
            fi

            echo "Navigating to: $TARGET_DIR"
            cd "$TARGET_DIR"

            chmod +x aws/deploy.sh
            ./aws/deploy.sh

4. ONE-TIME AWS EC2 SETUP COMMANDS (UBUNTU)
================================================================================

RECOMMENDED AWS SPECS:
  - Instance Type: t3.medium (2 vCPUs, 4 GiB RAM)
  - Storage: 20 GiB (gp3)
  - Note: No swap space is needed because t3.medium includes 4 GB of physical RAM!

Run these commands ONCE on your EC2 instance via SSH:

  # 1. Update system packages
  sudo apt-get update -y && sudo apt-get upgrade -y
  sudo apt-get install -y ca-certificates curl gnupg git

  # 2. Add Docker repository & GPG key
  sudo install -m 0755 -d /etc/apt/keyrings
  curl -fsSL https://download.docker.com/linux/ubuntu/gpg | sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg
  sudo chmod a+r /etc/apt/keyrings/docker.gpg

  echo \
    "deb [arch="$(dpkg --print-architecture)" signed-by=/etc/apt/keyrings/docker.gpg] https://download.docker.com/linux/ubuntu \
    "$(. /etc/os-release && echo "$VERSION_CODENAME")" stable" | \
    sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

  # 3. Install Docker Engine and Docker Compose
  sudo apt-get update -y
  sudo apt-get install -y docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin

  # 4. Enable Docker & grant permissions to ubuntu user
  sudo systemctl enable docker
  sudo systemctl start docker
  sudo usermod -aG docker $USER
  newgrp docker

  # 5. Clone repository to home directory
  cd ~
  git clone <YOUR_GITHUB_REPO_URL> MERN_Project
  cd MERN_Project

  # 6. Execute initial run
  chmod +x aws/deploy.sh
  ./aws/deploy.sh

5. GITHUB SECRETS CONFIGURATION (ONLY 2 SECRETS NEEDED)
================================================================================

Navigate to: GitHub Repo -> Settings -> Secrets and variables -> Actions -> New repository secret

Add only these 2 secrets:
  1. AWS_ACCESS_KEY_ID     : Your AWS Access Key ID (e.g., AKIA...)
  2. AWS_SECRET_ACCESS_KEY : Your AWS Secret Access Key

(Note: All SSH keys, security groups, EC2 t3.medium creation, and 20GB storage are
handled 100% automatically by GitHub Actions!)

================================================================================
6. HOW TO TRIGGER DEPLOYMENT
================================================================================

Whenever you commit and push to main:
  git add .
  git commit -m "feat: automated CI/CD pipeline"
  git push origin main

GitHub Actions will automatically connect via SSH, run `aws/deploy.sh`, rebuild 
the containers, and your app will be immediately live on:
  http://<YOUR_EC2_PUBLIC_IP>/
